cmake_minimum_required(VERSION 3.16)
project(DetectivesDossier CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra)
endif()

include_directories(src)
add_executable(game src/main.cpp)
add_executable(engine_tests tests/tests.cpp)
enable_testing()
target_compile_definitions(engine_tests PRIVATE ASSET_DIR="${CMAKE_SOURCE_DIR}/assets")
add_test(NAME engine_tests COMMAND engine_tests)

# ---- Qt GUI (built only if Qt 6 is found) ----
find_package(Qt6 QUIET COMPONENTS Widgets OPTIONAL_COMPONENTS Sql Multimedia)
if(Qt6_FOUND)
  set(CMAKE_AUTOMOC ON)
  add_executable(dossier_gui
    src/gui/main_gui.cpp
    src/gui/GameWidget.cpp
    src/gui/GameWidget.h)
  target_compile_definitions(dossier_gui PRIVATE ASSET_DIR="${CMAKE_SOURCE_DIR}/assets")
  target_link_libraries(dossier_gui PRIVATE Qt6::Widgets)
  # Password hashing test (needs only Qt Core): the last line must say "Auth tests passed."
  add_executable(auth_tests tests/auth_tests.cpp src/gui/Auth.cpp src/gui/Auth.h)
  target_link_libraries(auth_tests PRIVATE Qt6::Core)
  add_test(NAME auth_tests COMMAND auth_tests)
  # Phase 3 modules are optional so the game still builds on a laptop that lacks them.
  if(TARGET Qt6::Sql)
    target_link_libraries(dossier_gui PRIVATE Qt6::Sql)
    target_compile_definitions(dossier_gui PRIVATE HAVE_QT_SQL)
    # Log-in and saving (the game window asks the player to log in first)
    target_sources(dossier_gui PRIVATE src/gui/Database.cpp src/gui/Database.h src/gui/DbSql.h
      src/gui/Auth.cpp src/gui/Auth.h src/gui/AuthRules.h src/gui/LoginDialog.cpp src/gui/LoginDialog.h)
    # Database layer test (run it in Qt Creator; the last line must say "Database tests passed.")
    add_executable(db_tests tests/db_tests.cpp src/gui/Database.cpp src/gui/Database.h src/gui/DbSql.h)
    target_link_libraries(db_tests PRIVATE Qt6::Core Qt6::Sql)
    add_test(NAME db_tests COMMAND db_tests)
  else()
    message(STATUS "Qt6 Sql not found: install it with the Qt Maintenance Tool for saving and login")
  endif()
  if(TARGET Qt6::Multimedia)
    target_link_libraries(dossier_gui PRIVATE Qt6::Multimedia)
    target_compile_definitions(dossier_gui PRIVATE HAVE_QT_MULTIMEDIA)
    target_sources(dossier_gui PRIVATE src/gui/Audio.cpp src/gui/Audio.h src/gui/Sfx.h)
  else()
    message(STATUS "Qt6 Multimedia not found: install it with the Qt Maintenance Tool for sound")
  endif()
else()
  message(STATUS "Qt6 not found: building console game and tests only")
endif()
